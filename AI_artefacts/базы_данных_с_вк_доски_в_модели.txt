import enum
import uuid
from datetime import datetime

from sqlalchemy import Boolean, DateTime, Enum, Float, Integer, String, func
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class Base(DeclarativeBase):
    pass


class PhotoStatuses(str, enum.Enum):
    uploaded = 'uploaded'
    pending = "pending"
    processing = "processing"
    done = "done"
    failed = "failed"


class Photos(Base):
    __tablename__ = 'Photos'


    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    id_string: Mapped[str] = mapped_column(String(36), unique=True, nullable=False, index=True, default=lambda: str(uuid.uuid4()))
    
    object_key: Mapped[str] = mapped_column(string(500), nullable=False)
    preview_key = Mapped[str] = mapped_column(String(500), nullable=True)

    load_time: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())
    photo_size: Mapped[int] = mapped_column(Integer, nullable=False)

    status: Mapped[PhotoStatuses] = mapped_column(Enum(PhotoStatuses, native_enum=True), default=PhotoStatuses.uploaded, nullable=False)
    attempts: Mapped[int] = mapped_column(Integer, default=0, nullable=False)

    last_error_code: Mapped[Optional[int]] = mapped_column(Integer, nullable=True)
    last_error_message: Mapped[Optional[str]] = mapped_column(String(1000), nullable=True)

    is_private: Mapped[bool] = mapped_column(Boolean, default=False, nullable=False)
    
    owner_data_token: Mapped[Optional[str]] = mapped_column(
        String, 
        ForeignKey("tokens.data_token"), 
        nullable=True,
        index=True
    )

    duplicate_group_id: Mapped[Optional[int]] = mapped_column(
        Integer,

class PhotoAnalysis(Base):
    __tablename__ = "photo_analysis"

    # 1. Связь с фото (1 к 1)
    # photo_id является и Primary Key этой таблицы, и Foreign Key на photos.id
    # Это гарантирует, что у одного фото может быть только одна запись анализа
    photo_id: Mapped[int] = mapped_column(
        Integer, 
        ForeignKey("photos.id"), 
        primary_key=True
    )

    # 2. Метрики лиц (могут быть NULL до анализа)
    faces_count: Mapped[Optional[int]] = mapped_column(Integer, nullable=True)
    eyes_closed_count: Mapped[Optional[int]] = mapped_column(Integer, nullable=True)

    # 3. Метрики качества (могут быть NULL до анализа)
    is_blurred: Mapped[Optional[bool]] = mapped_column(Boolean, nullable=True)
    
    # Для значений 0-255 лучше использовать SmallInteger (экономит место), 
    # но Integer тоже ок. Главное - nullable=True
    blur_score: Mapped[Optional[int]] = mapped_column(Integer, nullable=True)
    quality_metric: Mapped[Optional[int]] = mapped_column(Integer, nullable=True)
    light_level: Mapped[Optional[int]] = mapped_column(Integer, nullable=True)

    # 4. Хэши для поиска дубликатов
    # perceptual_hash (pHash) - обычно 16-64 символа (hex)
    perceptual_hash: Mapped[Optional[str]] = mapped_column(String(64), nullable=True, index=True)
    
    # sha256_hash - всегда 64 символа (hex)
    sha256_hash: Mapped[Optional[str]] = mapped_column(String(64), nullable=True, index=True)

    # 5. Время завершения анализа
    # Заполняется ТОЛЬКО когда анализ успешно завершен
    analysis_at: Mapped[Optional[datetime]] = mapped_column(
        DateTime(timezone=True), 
        nullable=True
    )

    # --- Relationship ---
    # Позволяет писать: photo.analysis.faces_count
    # uselist=False означает связь "один к одному", вернется один объект, а не список
    photo = relationship("Photo", back_populates="analysis", uselist=False)
